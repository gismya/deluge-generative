# Perform's guards on the global MIDI commands RECORD, UNDO and PLAY with X held (#118, spec §19 F9 and F11).
# delugemu only (test_port_emu.py): it learns the commands in SETTINGS > MIDI > COMMANDS, which saves them to flash, so
# on a Deluge it would overwrite the player's own MIDI commands. Start as after boot: transport stopped, the current
# clip a synth or kit clip. Commands on MIDI channel 1: RECORD note 60, UNDO note 61, PLAY note 62.
expect playing=false
expect ui=instrument_clip

# SHIFT+SELECT opens SETTINGS; MIDI is its 5th item, COMMANDS the 7th in MIDI
button SHIFT down
wait 50
button 4 3 down
wait 50
button 4 3 up
wait 50
button SHIFT up
expect ui=sound_editor
encoder 3 4
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
encoder 3 6
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200

# COMMANDS: PLAY, RESTART, RECORD, TAP, UNDO. Learn each with LEARN held and a note in.
encoder 3 2
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
button LEARN down
wait 200
midi 90 3c 64
midi 80 3c 00
button LEARN up
wait 200
button BACK down
wait 50
button BACK up
wait 200
encoder 3 2
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
button LEARN down
wait 200
midi 90 3d 64
midi 80 3d 00
button LEARN up
wait 200
button BACK down
wait 50
button BACK up
wait 200
encoder 3 -4
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
button LEARN down
wait 200
midi 90 3e 64
midi 80 3e 00
button LEARN up
wait 200

# Out of SETTINGS
button BACK down

button BACK down
wait 50

button BACK down

button BACK up
wait 200

expect ui_mode=0

# The learned commands work in a stock view: MIDI RECORD turns recording on and off again
midi 90 3c 64
midi 80 3c 00
expect recording=normal
midi 90 3c 64
midi 80 3c 00
expect recording=off

# ... and MIDI UNDO reverts a gold-knob edit: the action moves from the undo log to the redo log
expect redo_log=0
encoder 4 3
wait 300
expect reversion=true
midi 90 3d 64
midi 80 3d 00
expect redo_log=1

include perform_open.txt
include perform_confirm.txt

# MIDI PLAY starts the Intro, as the panel button does
midi 90 3e 64
midi 80 3e 00
expect playing=true
expect event gen_section INTRO 0 20

# MIDI RECORD is a no-op until Capture (M4)
midi 90 3c 64
midi 80 3c 00
expect recording=off
expect playing=true

# MIDI PLAY with X held is the blocked restart: still playing, no fresh Intro
button X_ENC down
wait 100
midi 90 3e 64
midi 80 3e 00
button X_ENC up
expect playing=true
expect no_event gen_section INTRO 8

# MIDI UNDO after edits stock would log. Perform logged nothing, so there is nothing to revert either way: this step
# checks that UNDO changes and crashes nothing. The guard itself is reversion=false.
button 1 0 down
wait 50
button 1 0 up
encoder 4 3
encoder 2 1
wait 300
midi 90 3d 64
midi 80 3d 00
wait 500
expect undo_log=0
expect redo_log=0
expect reversion=false
expect root_ui=generative_perform
expect ui=generative_perform
expect playing=true

# MIDI PLAY stops; leaving (BACK held, LEAVE) lands in session view

expect playing=false
