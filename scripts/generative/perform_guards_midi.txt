# Perform's guards on the global MIDI commands RECORD, UNDO and PLAY with X held (#118, spec §19 F9 and F11).
# delugemu only (test_port_emu.py): it learns the commands in SETTINGS > MIDI > COMMANDS, which saves them to flash, so
# on a Deluge it would overwrite the player's own MIDI commands. Start as after boot: transport stopped, the current
# clip a synth or kit clip. Commands on MIDI channel 1: RECORD note 60, UNDO note 61, PLAY note 62.
expect playing=false
expect ui=instrument_clip

# SHIFT+SELECT opens SETTINGS; MIDI is its 5th item, COMMANDS the 7th in MIDI
button SHIFT down
wait 50
button 4 3 down
wait 50
button 4 3 up
wait 50
button SHIFT up
expect ui=sound_editor
encoder 3 4
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
encoder 3 6
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200

# COMMANDS: PLAY, RESTART, RECORD, TAP, UNDO. Learn each with LEARN held and a note in.

wait 200

wait 50
button 4 3 up
wait 200

wait 200
midi 90 3c 64
midi 80 3c 00
button LEARN up
wait 200
button BACK down
wait 50
button BACK up
wait 200
encoder 3 2
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
button LEARN down
wait 200
midi 90 3d 64
midi 80 3d 00
button LEARN up
wait 200
button BACK down
wait 50
button BACK up
wait 200
encoder 3 -4
wait 200
button 4 3 down
wait 50
button 4 3 up
wait 200
button LEARN down
wait 200
midi 90 3e 64
midi 80 3e 00
button LEARN up
wait 200
