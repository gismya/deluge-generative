# AFFECT ENTIRE on the CLIP page (#298, spec §13.5) over the sysex test port: it locks the part the page shows, or
# unlocks it; on SONG and KEYBOARD it only unlocks. Runs on a Deluge (deluge_probe.py session) or in delugemu
# (test_port_emu.py). Start as after boot: transport stopped, the current clip a synth or kit clip.
#
# Acid techno's parts: DRUMS row 1 (y 7), ACID row 2 (y 6), DRONE row 3, FX row 4. gen_focus part=<p> locked=<p>: the
# part on the gold knobs (-1: the Recipe level) and the part locked. gen_page_leds ... affect=1: AFFECT ENTIRE's LED,
# lit at the Recipe level. gen_edit part= lane=-1: a gold knob moved a part's param (lane -1: a Kit's kit level);
# gen_macro_knob: it steered a macro (F8, the Recipe level). gen_knobs page=<n> thin_drive=<0|1>: an F button pressed
# with a part focused, and whether the knobs are on its Thin / Drive set. gen_press freeze part=<p> and gen_press take
# part=<p> take=<t> y=<1|-1>: CROSS-SCREEN froze or thawed the focused part, and Y queued its next or previous take.
# Gold turns are fast: one edit per detent.
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view.h
# covers: src/deluge/gui/views/perform_view_internal.h src/deluge/gui/views/perform_view_clip.cpp
# covers: src/deluge/gui/views/perform_view_song.cpp src/deluge/gui/views/perform_view_knobs.cpp
# covers: src/deluge/generative/perform/clip_page.* src/deluge/generative/perform/knob_page.h
# covers: src/deluge/generative/perform/sound_mapping.* src/deluge/generative/firmware/sound_controls.*
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/gui/menu_item/generative/perform_menu.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
start_state
include perform_open.txt
include perform_confirm.txt

# ---- Nothing focused since entry: the CLIP page shows no part, and AFFECT ENTIRE does nothing ----
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect no_event gen_focus within=1

# ---- PLAY; hold ACID's row on the SONG page and let go: the CLIP page shows ACID, the knobs are at the Recipe level ----
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_controls sets=13
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 3 6 down
expect event gen_focus part=1 locked=-1
pad 3 6 up
expect event gen_focus part=-1 locked=-1
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
expect event gen_page_leds song=0 clip=1 keyboard=0 affect=1

# ---- AFFECT ENTIRE locks ACID: its LED goes dark, and the knobs, CROSS-SCREEN and Y act on ACID ----
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=1 locked=1
expect event gen_page_leds song=0 clip=1 keyboard=0 affect=0
button F2 down
wait 50
button F2 up
expect event gen_knobs page=2 thin_drive=0
encoder 4 1 fast
expect next event gen_edit part=1 lane=-1 control=BRIGHTNESS
button CROSS_SCREEN_EDIT down
wait 50
button CROSS_SCREEN_EDIT up
expect event gen_press freeze part=1
button CROSS_SCREEN_EDIT down
wait 50
button CROSS_SCREEN_EDIT up
expect event gen_press freeze part=1
encoder 0 1
expect event gen_press take part=1 take=1 y=1

# ---- Again: it unlocks, LED lit, the knobs on F8's macros; the page still shows ACID, so a third press locks it ----
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=-1 locked=-1
expect event gen_page_leds song=0 clip=1 keyboard=0 affect=1
button F8 down
wait 50
button F8 up
encoder 4 1 fast
expect event gen_macro_knob macro=0
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=1 locked=1

# ---- ACID locked, DRUMS' row held and let go: the knobs are back on ACID, the CLIP page shows DRUMS ----
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 3 7 down
expect event gen_focus part=0 locked=1
pad 3 7 up
expect event gen_focus part=1 locked=1
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
# AFFECT ENTIRE locks DRUMS: the knobs move from ACID to the Kit's kit-level pages
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=0 locked=0
button F2 down
wait 50
button F2 up
expect event gen_knobs page=2 thin_drive=0
encoder 4 1 fast
expect next event gen_edit part=0 lane=-1 control=BRIGHTNESS

# ---- The lock lasts across SONG and KEYBOARD; there AFFECT ENTIRE only unlocks ----
button KEYBOARD down
wait 50
button KEYBOARD up
expect event gen_page KEYBOARD
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
expect no_event gen_focus within=1
encoder 4 1 fast
expect next event gen_edit part=0 lane=-1 control=BRIGHTNESS
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=-1 locked=-1
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect no_event gen_focus within=1
# Locked again on CLIP, then unlocked on KEYBOARD; there, a second press does nothing
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=0 locked=0
button KEYBOARD down
wait 50
button KEYBOARD up
expect event gen_page KEYBOARD
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=-1 locked=-1
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect no_event gen_focus within=1

# ---- Inside the sound editor (SOUND EDITOR on the CLIP page) AFFECT ENTIRE does nothing (spec §13.7) ----
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
pad 2 6 down

pad 2 6 up
expect event gen_editor open part=0
expect ui=sound_editor

button AFFECT_ENTIRE up
expect no_event gen_focus within=1
button BACK down

expect ui=generative_perform

# ---- A lock made on the CLIP page is dropped by a stop ----
button AFFECT_ENTIRE down
wait 50

expect event gen_focus part=0 locked=0
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event gen_focus part=-1 locked=-1

# ---- A SONG row held into the CLIP page: AFFECT ENTIRE locks its part and ends the hold, so letting go of the row
# changes nothing (spec §13.5) ----

wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 3 6 down
expect event gen_focus part=1 locked=-1
button CLIP_VIEW down

button CLIP_VIEW up
expect event gen_page CLIP

wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=1 locked=1
pad 3 6 up
expect no_event gen_focus within=1

# ---- Back on the CLIP page while locked, it still shows the locked part: AFFECT ENTIRE unlocks it ----
button SESSION_VIEW down
wait 50
button SESSION_VIEW up

button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP

# ---- A lock on the CLIP page ends a double tap begun on SONG: a row tap, CLIP, AFFECT ENTIRE, SONG and a tap of the
# same row, all within 0.35 s, doesn't unlock the part (no expects between, so the taps stay that close). DRUMS is
# locked first, so a release never takes the knobs to the Recipe level: locked=-1 would be the unlock ----
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 3 7 down
wait 20
pad 3 7 up
pad 3 7 down
wait 20
pad 3 7 up
expect event gen_focus part=0 locked=0
wait 500
mark

pad 3 6 up
button CLIP_VIEW down
wait 20
button CLIP_VIEW up
button AFFECT_ENTIRE down
wait 20

button SESSION_VIEW down

button SESSION_VIEW up

expect event gen_focus part=1 locked=1 since=mark within=3
expect no_event /gen_focus part=-?[0-9]+ locked=-1/ since=mark within=2
# Still locked: AFFECT ENTIRE on SONG unlocks it
button AFFECT_ENTIRE down
wait 50
button AFFECT_ENTIRE up
expect event gen_focus part=-1 locked=-1
