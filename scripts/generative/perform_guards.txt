# Perform's guards on RECORD, X+PLAY and undo (#118, #413, spec §19 F9 and F11) over the sysex test port. Runs on a Deluge
# (deluge_probe.py run) or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current clip a
# synth or kit clip.
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view.h
# covers: src/deluge/gui/views/perform_view_internal.h src/deluge/gui/views/session_view.cpp
# covers: src/deluge/gui/ui/generative_* src/deluge/hid/buttons.cpp src/deluge/generative/firmware/entry_leave.cpp
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/generative/firmware/output_sink.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/generative/firmware/perform_adapter.cpp
# covers: src/deluge/generative/firmware/restart_began.cpp
start_state
include perform_open.txt

# The test port's load song (test builds only) refuses while something is open over the root UI, here DISCARD SONG?
load song SONGS/NO_SUCH_SONG.XML
expect event song_load_failed path=SONGS/NO_SUCH_SONG.XML why=busy
expect ui=context_menu

# RECORD armed in the stock song at DISCARD SONG? would stay on in Perform, which can't turn it off: entry disarms it
button RECORD down
wait 50
button RECORD up
expect recording=normal
include perform_confirm.txt
expect recording=off

# ...and in Perform, whose song performAdapter still uses
load song SONGS/NO_SUCH_SONG.XML
expect event song_load_failed path=SONGS/NO_SUCH_SONG.XML why=busy
expect root_ui=generative_perform

# Perform keeps the undo log empty from entry, and nothing may revert (spec §12.9, F11)
expect undo_log=0
expect redo_log=0
expect reversion=false

# RECORD held + PLAY starts playback only: no output recording, which in Perform is SHIFT+RECORD alone (spec §13.5)
button RECORD down
wait 100
button PLAY down
wait 50
button PLAY up
wait 100
button RECORD up
expect playing=true
expect resampling=false
expect recording=off
expect event gen_section INTRO 0 within=20

# RECORD is a no-op until Capture (M4): recording stays off, so no automation lands in Perform's clips (F9)
button RECORD down
wait 50
button RECORD up
expect recording=off
expect resampling=false
expect playing=true

# X+PLAY is Perform's restart (spec §13.2, perform_restart.txt): still playing, the piece starts again from its Intro
# on the next bar line, rather than stock's restart of the song
button X_ENC down
wait 100
button PLAY down
wait 50
button PLAY up
wait 100
button X_ENC up
expect playing=true
expect event gen_restart as_it_is within=15
expect event gen_section INTRO 0 clip_pos=0 bar=0 within=5

# Edits that stock logs as undoable actions log nothing in Perform: a song-FX gold knob (the F1 page) and a tempo turn
button F1 down
wait 50
button F1 up
encoder 4 3
encoder 4 -3
encoder 2 1
encoder 2 -1
wait 300
expect undo_log=0
expect redo_log=0
expect reversion=false
expect recording=off

button PLAY down
wait 50
button PLAY up
expect playing=false

# From stop, X held + PLAY starts the piece from its Intro, as PLAY alone does: Perform never starts mid-piece
button X_ENC down
wait 100
button PLAY down
wait 50
button PLAY up
wait 100
button X_ENC up
expect playing=true
expect event gen_section INTRO 0 within=20
button PLAY down
wait 50
button PLAY up
expect playing=false

# Leaving (BACK held, LEAVE) lands in session view, where undo works again
include perform_leave.txt
expect ui_mode=0
expect reversion=true
expect recording=off
expect resampling=false
