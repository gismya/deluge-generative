# House's hardware session (#132 decision 9, spec §20 H1 and H16): Acid techno changes into House by BUILD INTO A DROP,
# its 8- or 16-bar Build gliding from 125-135 to House's 120-126 BPM onto House's Drop. House then plays its Groove
# (claps on 2 and 4, the offbeat bass, the stabs and Strings, the Lead answering in each bar's second half); a forced
# Break keeps the Stab, the Strings and the closed hat; a forced Groove follows a phrase later, and a forced Build (4 or
# 8 bars, without the kick and the Bass) rolls the Snare up to 32nds through the 1-beat Gap into a second Drop. gen_h1 (step
# decisions) and gen_h16 (culls, the last culled voice's part, voices, the slowest render) report every 4 bars (about
# 8 s); the runner reads them over the Groove and the second Drop, and stops if H1's mean passes 150 us or culled= ever
# names DRUMS (a kick) or BASS. gen_mem: SDRAM free before entry, when Acid's roster is freed, and after leaving.
# Runs on a Deluge (deluge_probe.py session) or in delugemu (test_port_emu.py). Start as after boot: transport
# stopped, the current clip a synth or kit clip.
# gen_transition start tempo_off=: the song's tempo against House's Recipe's at its first step's decision tick (12 ticks
# early), in hundredths of a BPM: under 10 BPM.
# Pinned rolls (hold entropy): the same Recipes and seeds every run.
# needs: a Deluge for its timings, and about 4 minutes in delugemu, so check_all.py skips it; perform_genre_change_house.txt is the functional check
# For figures: --events 'gen_confirm|gen_transition|gen_section|gen_h1|gen_h16|gen_mem'.
# covers: src/deluge/generative/core/genres/house.cpp src/deluge/generative/core/genres/house_palette.cpp
# covers: src/deluge/generative/core/generator_patterns.cpp src/deluge/generative/core/generator_sections.cpp
# covers: src/deluge/generative/firmware/adapter_timing.* src/deluge/generative/firmware/output_sink.cpp
# covers: src/deluge/generative/firmware/perform_adapter.* src/deluge/generative/firmware/entry_leave.cpp
# covers: src/deluge/gui/context_menu/generative.* src/deluge/gui/views/perform_view.cpp
# covers: src/deluge/generative/firmware/transitions.* src/deluge/generative/perform/transition_lifecycle.h
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/generative/firmware/retirements.cpp src/deluge/generative/firmware/sound_swaps.cpp
# covers: src/deluge/gui/menu_item/generative/perform_menu.cpp src/deluge/gui/views/perform_view_clip.cpp
# covers: src/deluge/gui/views/perform_view_song.cpp
start_state
hold entropy
mark now
include perform_open.txt
include perform_confirm.txt
expect event /^gen_mem before_entry internal=\d+ external=\d+/ since=mark
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0

# ---- HOUSE -> BUILD INTO A DROP: Acid's Build glides to House's tempo, onto House's Drop ----
include perform_transition_menu.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre house
expect event gen_genre_pick house row=HOUSE
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect ui=context_menu
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_confirm house style=0 result=0
expect event gen_transition_preload done error=0 parts=6 within=30
expect event gen_section BUILD within=60
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]/ within=20
expect event /^gen_transition start style=0 us=\d+ tempo_off=\d{1,3}$/ within=60
mark now
expect event /^gen_outputs_freed outputs=\d+ total=4$/ since=mark within=40
expect event /^gen_mem after_free internal=\d+ external=\d+/ since=mark within=40

# ---- The Drop and the Groove: H1 and H16 every 4 bars, a phrase and more ----
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20

# ---- A forced Break on the next phrase line: the Stab, the Strings and the closed hat ----
expect event gen_phrase bar= within=40
pad 17 6 down
wait 50
pad 17 6 up
expect event gen_press force BREAK
expect event gen_section BREAK within=40
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
pad 17 7 down
wait 50
pad 17 7 up
expect event gen_press force GROOVE
expect event gen_section GROOVE within=40

# ---- A forced Build: no kick or Bass, the Snare rolling to 32nds, a beat's Gap, then the second Drop ----
pad 17 5 down
wait 50
pad 17 5 up
expect event gen_press force BUILD
expect event gen_section BUILD within=40
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect event gen_section GROOVE within=40
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect event gen_h1 steps= within=20
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]\d* render_max_us=\d+/ within=20
expect playing=true

button PLAY down
wait 50
button PLAY up
expect playing=false
mark now
include perform_leave.txt
expect ui_mode=0
expect event /^gen_mem after_leave internal=\d+ external=\d+/ since=mark within=10
release entropy
