# MIDI in Perform (#117, spec §13.7, §19 F8, F10, F18) over the sysex test port. Runs on a Deluge (deluge_probe.py
# session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current clip a synth or kit
# clip.
#
# "hold midi_setup" sets, in RAM only: MIDI follow A on channel 16 (status nibble F) with feedback at the HIGH rate,
# follow's Tracks 1-3 on channels 13-11 (C, B, A), the global commands on channel 15 (E: PLAYBACK_RESTART note 60, PLAY 61, LOOP 64,
# UNDO 66, NEXT_SONG 70) and TRANSPOSE on all of channel 14 (D). Acid's parts: DRUMS row 1 (y 7), ACID row 2 (y 6).
# Events: gen_macro_midi macro= units= (0 Energy, 3200 = 50 points); gen_midi_command transpose|restart blocked|<n>
# ignored|<n> swallowed (n: GlobalMIDICommand; swallowed: by a Generative job's modal screen); gen_key_set root= scale= stopped|queued; gen_edit: a param edit moved a home;
# gen_midi_feedback ccs= ms= bytes_per_s= peak_pass=: each second's live-value feedback, H17 (DIN bytes, 3 a CC);
# gen_midi_feedback_edit cc= value=: a player edit's feedback, sent at once.
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view.h
# covers: src/deluge/gui/views/perform_view_internal.h src/deluge/gui/views/perform_view_knobs.cpp
# covers: src/deluge/generative/firmware/perform_midi.* src/deluge/generative/perform/midi_in.*
# covers: src/deluge/generative/firmware/sound_controls.* src/deluge/generative/firmware/entry_leave.cpp
# covers: src/deluge/io/midi/midi_follow.cpp
# covers: src/deluge/gui/views/perform_view_clip.cpp src/deluge/generative/firmware/perform_adapter.cpp
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/generative/firmware/output_sink.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/generative/perform/clip_page.* src/deluge/generative/perform/perform_steering.h
# covers: src/deluge/gui/l10n/english.json src/deluge/gui/l10n/g_english.cpp
start_state
hold midi_setup
include perform_open.txt

# ---- Entry's modal screen swallows a global command's note-on (F10): MIDI PLAY with entry held at its first
# boundary starts nothing, before or after entry ----
# OK by hand rather than include perform_confirm.txt: that one waits for gen_perform_entered, and the MIDI PLAY has to
# land while entry is still held.
hold entry
button 4 3 down
wait 50
button 4 3 up
expect event gen_entry_held within=10
midi 9e 3d 64
expect event gen_midi_command 1 swallowed
midi 8e 3d 00
release entry
expect event gen_perform_entered within=20
expect root_ui=generative_perform within=10
expect ui=generative_perform
expect ui_mode=0
expect playing=false

# ---- Stopped: commands and macros ----
# TRANSPOSE: note 62 (D) sets the key's root at once while stopped; its note-off does nothing
midi 9d 3e 64
expect event gen_key_set root=2
expect event gen_midi_command transpose note=62
midi 8d 3e 00
expect no_event gen_key_set within=1

# NEXT_SONG, LOOP and UNDO are ignored (their gen_midi_command events): still in Perform, still stopped. Perform logs
# no undo steps (a tempo turn included), so the undo log stays empty and an UNDO would find nothing either: the
# ignored event is the check that it never reached stock's UNDO.
encoder 2 1
wait 300
keep bpm
midi 9e 46 64
midi 8e 46 00
expect event gen_midi_command 10 ignored
midi 9e 40 64
midi 8e 40 00
expect event gen_midi_command 4 ignored
midi 9e 42 64
midi 8e 42 00
expect event gen_midi_command 6 ignored
wait 500
expect root_ui=generative_perform
expect ui=generative_perform
expect playing=false
expect undo_log=0
expect bpm=kept

# CCs 114-117 on the follow channel steer the macros' centres, 0-127 over 0-50
midi bf 72 7f
expect event gen_macro_midi macro=0 units=3200
midi bf 73 00
expect event gen_macro_midi macro=1 units=0
midi bf 75 40
expect event gen_macro_midi macro=3 units=1613
# ...on no other channel
midi b0 72 10
expect no_event gen_macro_midi within=1

# ---- MIDI PLAY starts the Intro ----
midi 9e 3d 64
midi 8e 3d 00
expect playing=true
expect event gen_section INTRO 0 within=20

# PLAYBACK_RESTART is blocked: still playing, no fresh Intro
midi 9e 3c 64
midi 8e 3c 00
expect event gen_midi_command restart blocked
expect playing=true
expect no_event gen_section INTRO within=8

# TRANSPOSE while playing queues the key change to the next phrase line
midi 9d 43 64
midi 8d 43 00
expect event gen_key_set root=7
expect event gen_midi_command transpose note=67

# A macro CC while playing lands in the core
midi bf 72 00
expect event gen_macro_midi macro=0 units=0
expect event gen_macro_landed macro=0 units=0 within=5

# ---- MIDI follow goes where the gold knobs point ----
# Nothing focused: CC 74 on the follow channel edits the song's LPF, a param no sound control moves
midi bf 4a 30
expect event gen_edit control=none
# ACID focused (row 2 double-tapped, locked): CC 74 on the follow channel moves its Brightness home
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
midi bf 4a 10
expect event gen_edit part=1 lane=-1 control=BRIGHTNESS
# The live values go out as feedback while it plays (H17): every one-second window under 1000 B (a third of DIN's
# 3125 B/s link; hardware measured 361-600, delugemu 431-568), checked once the stress and focus windows are in too.
# The rate is checked on the Deluge only (hardware_only, #417): a slowed delugemu sends its windows in bursts
mark
expect event /^gen_midi_feedback ccs=[1-9]\d* / within=5
# ...and at their worst, every sound-control param ramping every step
hold stress
sleep 8
release stress
expect event /^gen_midi_feedback ccs=[1-9]\d* / within=5
# A follow CC isn't sent back to the controller. A piece started with Perform's ramps held moves no param (a hold
# taken mid-piece leaves the ramps already running), so once the focus's full pass has gone out, after CC 74 lands
# no pass sends anything
midi 9e 3d 64
midi 8e 3d 00
expect playing=false
hold ramps
midi 9e 3d 64
midi 8e 3d 00
expect playing=true
expect event gen_section INTRO 0 within=20
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
expect event /^gen_midi_feedback ccs=[1-9]\d* / within=5
hardware_only expect no_event /^gen_midi_feedback .* bytes_per_s=\d{4,} / since=mark within=1
sleep 3
midi bf 4a 50
expect event gen_edit part=1 lane=-1 control=BRIGHTNESS
expect no_event gen_midi_feedback within=3
release ramps

# ---- External notes reach no part's Output (F18) ----
# Stopped, so any voice is one a note started. Each Track channel would play a part's Output; follow A the focused one
midi 9e 3d 64
midi 8e 3d 00
expect playing=false
expect voices=0 within=20
# Stopped (no passes): a CLIP page fader tap sends no feedback while the knobs are on the song (a stop drops focus),
# since ACID's value would go out on the song's CC
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
pad 3 5 down
wait 50
pad 3 5 up
expect event gen_faders part=1
expect no_event gen_midi_feedback_edit within=1
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
midi 9c 3c 7f
midi 9c 24 7f
midi 9b 3c 7f
midi 9b 24 7f
midi 9a 3c 7f
midi 9a 24 7f
midi 9f 40 7f
midi ec 00 70
midi bc 01 7f
midi dc 7f
wait 500
expect voices=0
midi 8c 3c 00
midi 8c 24 00
midi 8b 3c 00
midi 8b 24 00
midi 8a 3c 00
midi 8a 24 00
midi 8f 40 00
midi ec 00 40
midi bc 01 00

# ...and with ACID focused, a fader tap (Brightness, CC 74) sends its live value once
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
pad 5 5 down
wait 50
pad 5 5 up
expect event gen_midi_feedback_edit cc=74
expect no_event gen_midi_feedback_edit within=1

# ---- Leave: release puts the player's MIDI settings back ----
include perform_leave.txt
release midi_setup
