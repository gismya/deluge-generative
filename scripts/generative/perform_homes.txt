# Sound-control homes and player edits (#112, spec §11.6, §13.5) over the sysex test port. Runs on a Deluge
# (deluge_probe.py session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current
# clip a synth or kit clip.
#
# Acid techno's parts: DRUMS row 1 (y 7), ACID row 2 (y 6), DRONE row 3, FX row 4. The CLIP page's faders, from spec row
# 3 (y 5) down: Brightness, Thin, Decay, Space, Drive, Wobble. gen_faders part=<p>, then each fader in that order:
# home/live pads (1–16) where they differ, home= where the live value is on home's pad, or - where the part's preset
# doesn't use the control. at_home follows part= when every live value is on its home's pad. gen_edit: a player edit
# while playing, with the control whose home it moved, or control=none. gen_knobs: a focused part's F page, and whether
# F7 again put Thin / Drive on it. gen_rings ok|stale upper= lower=: on Thin / Drive, whether the knob rings show them
# (stale: stock's levels came back). gen_macro_knob macro=: a gold knob steered a macro (0 Energy). Gold turns are fast
# (one edit per detent), so each turn sends one gen_edit and "expect next event" checks that one.
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view.h
# covers: src/deluge/gui/views/perform_view_internal.h src/deluge/gui/views/perform_view_knobs.cpp
# covers: src/deluge/gui/views/perform_view_keyboard.cpp src/deluge/gui/views/perform_view_clip.cpp
# covers: src/deluge/generative/perform/knob_page.h src/deluge/generative/perform/sound_mapping.*
# covers: src/deluge/generative/perform/perform_steering.h src/deluge/generative/firmware/sound_controls.*
# covers: src/deluge/generative/core/generator_controls.cpp src/deluge/gui/ui/ui.cpp src/deluge/hid/encoder_input.cpp
# covers: src/deluge/generative/firmware/entry_leave.cpp src/deluge/gui/context_menu/generative.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/perform/clip_page.*
# covers: src/deluge/gui/l10n/english.json src/deluge/gui/l10n/g_english.cpp
start_state
include perform_open.txt
include perform_confirm.txt

# ---- The CLIP page, stopped ----
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP

# ---- PLAY: the Intro shades the sound, so the layers sit away from home ----
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_controls sets=13
# Under the leave menu, F8 keeps the macros (F16): the upper knob steers Energy, not song-level decimation
include perform_hold_back.txt
button F8 down
wait 50
button F8 up
encoder 4 1 fast
expect event gen_macro_knob macro=0
button BACK down
wait 50
button BACK up
expect ui=generative_perform
# Lock ACID from the SONG page (double tap its row); the CLIP page then shows it
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP

# ---- A fader tap sets home: pad 12 on Brightness's row. The Intro's darker layer rides on it, off home's pad ----
pad 11 5 down
wait 50
pad 11 5 up
expect event gen_edit part=1 lane=-1 control=BRIGHTNESS
expect event gen_faders part=1 12/

# ---- The focused part's stock pages: a param a control moves sets its home (spec §13.5) ----
# CUTOFF (F2, upper knob) = Brightness
button F2 down
wait 50
button F2 up
expect event gen_knobs page=2 thin_drive=0
encoder 4 1 fast
expect event gen_edit part=1 lane=-1 control=BRIGHTNESS
# RELEASE (F3, lower knob) = Decay
button F3 down
wait 50
button F3 up
encoder 5 1 fast
expect event gen_edit part=1 lane=-1 control=DECAY
# REVERB (F5, lower knob) = Space
button F5 down
wait 50
button F5 up
encoder 5 1 fast
expect event gen_edit part=1 lane=-1 control=SPACE
# DELAY AMOUNT (F4, lower knob) = Space only where the preset uses the delay, and MOD DEPTH (F6, lower knob) = Wobble
# only where it has the LFO1 -> pitch cable. ACID's preset is rolled from its palette, and some of its picks have one
# and some don't, so each turn only checks that it reaches F3: the event names SPACE / WOBBLE, or none.
button F4 down
wait 50
button F4 up
encoder 5 1 fast
expect event gen_edit
button F6 down
wait 50
button F6 up
encoder 5 1 fast
expect event gen_edit
# A param no control moves (RESONANCE, F2 lower) moves no home
button F2 down
wait 50
button F2 up
encoder 5 1 fast
expect next event gen_edit control=none

# ---- F7 alone stays stutter / CUSTOM 1; F7 again is Thin / Drive; again, back ----
button F7 down
wait 50
button F7 up
expect event gen_knobs page=7 thin_drive=0
encoder 5 1 fast
expect next event gen_edit control=none
button F7 down
wait 50
button F7 up
expect event gen_knobs page=7 thin_drive=1
expect event gen_rings ok
encoder 4 1 fast
expect event gen_edit part=1 lane=-1 control=THIN
encoder 5 1 fast
expect event gen_edit part=1 lane=-1 control=DRIVE
# The rings stay on Thin and Drive after stock's 25 ms refresh of each write (F15)
wait 300
expect no_event gen_rings stale within=1
expect event gen_rings ok
# Under an overlay (the leave menu) the gold knobs and F buttons stay Perform's (F16): the knob still turns Thin
include perform_hold_back.txt
encoder 4 1 fast
expect event gen_edit part=1 lane=-1 control=THIN
wait 300
expect no_event gen_rings stale within=1
# F7 pressed inside the menu is the third press: Thin / Drive ends, and stays ended once BACK closes the menu
button F7 down
wait 50
button F7 up
expect event gen_knobs page=7 thin_drive=0
button BACK down
wait 50
button BACK up
expect ui=generative_perform
encoder 4 1 fast
expect next event gen_edit control=none
expect no_event gen_rings ok within=1

# A fader tap sets Brightness's home exactly, on pad 12, whatever drift is playing (a knob turn moves it by the amount
# turned less the offset playing then, which drift moves, #107): the stop below checks this home survives
pad 11 5 down
wait 50
pad 11 5 up
expect event gen_edit part=1 lane=-1 control=BRIGHTNESS

# ---- PLAY stops: focus and Thin / Drive go; every param goes back to its home, which the edits moved ----
button F7 down
wait 50
button F7 up
expect event gen_knobs page=7 thin_drive=1
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event gen_knobs thin_drive=0 refocus
# Every fader's live value is back on its home's pad, and Brightness's is the home the tap set
expect event gen_faders part=1 at_home 12=

# ---- PLAY again, ACID locked again: its F7 is stutter / CUSTOM 1, not Thin / Drive ----
button PLAY down
wait 50
button PLAY up
expect playing=true
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
encoder 4 1 fast
expect next event gen_edit control=none

# ---- Focus moving away ends Thin / Drive: the part comes back on its stock F7 ----
button F7 down
wait 50
button F7 up
expect event gen_knobs page=7 thin_drive=1
# Hold DRUMS (the lock stays, so ACID is focused again on release)
pad 9 7 down
expect event gen_knobs thin_drive=0 refocus
expect event gen_focus part=0 locked=1
pad 9 7 up
expect event gen_focus part=1 locked=1
encoder 4 1 fast
expect next event gen_edit control=none

# ---- The leave menu opening over a held row lets go of the hold; the lock stays (spec §13.5) ----
# Past the double-tap window since DRUMS' last press: the fast turn above takes no time
wait 400
pad 9 7 down
expect event gen_focus part=0 locked=1
mark
include perform_hold_back.txt
# Let go before the menu's event, as the menu opens
expect event gen_focus part=1 locked=1 since=mark
pad 9 7 up
button BACK down
wait 50
button BACK up
expect ui=generative_perform

# ---- A row held on SONG and let go on CLIP: the hold ends on release; CLIP shows the part held last (DRUMS) ----
pad 9 7 down
expect event gen_focus part=0 locked=1
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
pad 9 7 up
expect event gen_focus part=1 locked=1
pad 11 5 down
wait 50
pad 11 5 up
expect event gen_edit part=0 lane=-1 control=BRIGHTNESS

# ---- A cable an edit deletes moves no home (#176 r2) ----
# MOD DEPTH (F6, lower knob) all the way up: ACID's LFO1 -> pitch cable at the top (knob 64), made if the preset has
# none. A cable's knob runs -192..64 with amount 0 at -64, so 32 detents (128 edges) reach the top from any amount >= 0.
button F6 down
wait 50
button F6 up
encoder 5 32 fast
button PLAY down
wait 50
button PLAY up
expect playing=false within=40
# The next piece binds the cable to Wobble; turned back down to 0, stock deletes it inside the edit. Motion 0 first (the
# KEYBOARD page's Motion fader, bottom row): drift scales the cable too (#107), and 32 detents must land on 0
button KEYBOARD down
wait 50
button KEYBOARD up
expect event gen_page KEYBOARD
pad 6 0 down
wait 50
pad 6 0 up
expect event gen_macro_pad macro=3 keep units=0
button PLAY down
wait 50
button PLAY up
expect playing=true
button SESSION_VIEW down
wait 50
button SESSION_VIEW up
expect event gen_page SONG
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
# Slow: one step an edit, as stock deletes the cable inside the edit that brings it to 0; a fast turn's 4-step edits
# end on the same amount but leave the cable in place
encoder 5 -32
# The 32 slow detents queue about 21 s, and the expect waits for the queue first
expect event gen_edit part=1 lane=-1 control=WOBBLE within=25
expect event gen_edit part=1 control=WOBBLE gone within=40
button PLAY down
wait 50
button PLAY up
expect playing=false

# Holding BACK opens LEAVE PERFORM?; LEAVE keeps nothing (spec §13.8)
include perform_leave.txt
