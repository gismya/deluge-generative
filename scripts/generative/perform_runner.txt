# The runner, Perform entry and leaving (#105, #119) over the sysex test port. Runs on a Deluge (deluge_probe.py run)
# or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current clip a synth or kit clip.
# Entry: SHIFT+SONG → GENERATIVE → PERFORM → PERFORM FROM → GENRE → TECHNO → DISCARD SONG? → OK (spec §14.1, §14.3).
# Test events to read from the log: gen_entry done|cancelled|failed ran=1|0 ms= (H9; ran=0: cancelled before its
# task ran), gen_left ms= delete_us=, and gen_mem before_entry /
# after_leave (memory after leaving matches memory before entry).
start_state
expect playing=false

# SHIFT+SONG from the clip goes to Song view
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=session

# ---- 1. BACK during entry cancels it: the player's song plays on, untouched (spec §14.3) ----
button PLAY down
wait 50
button PLAY up
expect playing=true
# SHIFT+SONG right after PLAY is sometimes missed in stock session view; give it a moment
sleep 1
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=context_menu
button 4 3 down
wait 50
button 4 3 up
expect ui=sound_editor
button 4 3 down
wait 50
button 4 3 up
wait 300
button 4 3 down
wait 50
button 4 3 up
expect ui=context_menu
# OK with entry held at its first boundary (a test-port hook: the job waits there until released or cancelled), then
# BACK. A timed BACK raced the build: a 5 ms gap failed 12/60 in delugemu when the build finished first.
hold entry
button 4 3 down
wait 50
button 4 3 up
expect event gen_entry_held within=10
button BACK down
wait 50
button BACK up
expect event gen_entry cancelled ran=1 within=10
expect root_ui=session
expect ui=session
expect ui_mode=0
expect playing=true
button PLAY down
wait 50
button PLAY up
expect playing=false

# ---- 2. Entry, then the leave menu on BACK hold while Perform plays (spec §13.8) ----

expect ui=context_menu
button 4 3 down

button 4 3 down
wait 50
button 4 3 up
wait 300
button 4 3 down
wait 50

expect ui=context_menu
button 4 3 down
wait 50
button 4 3 up
expect event gen_entry done within=20
expect event gen_perform_entered within=10
expect root_ui=generative_perform within=10
expect ui=generative_perform
expect ui_mode=0
expect playing=false

button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0

# A BACK tap with nothing queued cancels nothing, and Perform plays on
button BACK down
wait 50
button BACK up
wait 300
expect ui=generative_perform
expect playing=true

# Hold BACK: LEAVE PERFORM? over Perform, which plays on. CANCEL goes back.
button BACK down
wait 700

expect ui=context_menu
expect root_ui=generative_perform
expect playing=true
encoder 3 1
wait 100
button 4 3 down
wait 50
button 4 3 up

expect event gen_perform_left within=5
