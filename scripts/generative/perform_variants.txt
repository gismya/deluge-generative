# Perform's VARIANTS ▸ and a live variant switch landing on a Drop (#495, spec §3.6, §13.6, ADR 0008 "Drop changes and
# live steering") over the sysex test port: Acid techno shows no VARIANTS row; Acid cuts to ROOTS DUB, whose Perform
# menu has VARIANTS ▸ after REGENERATE, the Regenerate page's rows on the playing piece's ticks (SKANK and DRUMS on
# RANDOM). DRUMS opens its multi-select: unticking ONE DROP, then ROCKERS, queues the ticks for the next Drop (VAR in
# the status line), and a forced 1-bar Build lands them on the Drop: ONE DROP isn't ticked, so the Drop draws within
# the ticks, STEPPERS, and the instance applies its genre copy again. gen_h1 reports around that Drop (H1, spec §20).
# Runs on a Deluge (deluge_probe.py session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped,
# the current clip a synth or kit clip.
#
# gen_variants_menu open sets=N, then gen_variants_row <SET> <shows> per set (Random, a variant's OLED name, n of m), as
# VARIANTS ▸ opens. gen_variants_press set=<name> allowed=<mask> outcome=<0 accepted>: a tick turned, queued;
# gen_variants set=<name> allowed=<mask> shows=<text>: the row's ticks after it. gen_variants_landed step=N drop=<1 on a
# Drop's first step> <set>=<variant>:<allowed> ...: the picks the piece plays moved on step N. Pinned rolls (hold
# entropy): the Dub Recipe plays ONE DROP.
# For figures: --events 'gen_variants|gen_variants_press|gen_variants_landed|gen_section|gen_h1|gen_h16'.
# covers: src/deluge/gui/menu_item/generative/perform_menu.* src/deluge/generative/perform/perform_steering.h
# covers: src/deluge/generative/perform/queue_text.* src/deluge/generative/core/variants.*
# covers: src/deluge/generative/core/generator_steering.cpp src/deluge/generative/core/genres/dub.cpp
# covers: src/deluge/generative/core/steering.h src/deluge/generative/firmware/perform_adapter.cpp
# covers: src/deluge/gui/l10n/english.json src/deluge/gui/l10n/g_english.cpp
# covers: src/deluge/generative/core/types.h src/deluge/generative/firmware/output_sink.cpp
# covers: src/deluge/generative/firmware/transitions.* src/deluge/generative/perform/transition_lifecycle.h
# covers: src/deluge/generative/perform/oled_page.* src/deluge/generative/perform/text.*
# covers: src/deluge/gui/menu_item/generative/regenerate_page.* src/deluge/gui/views/perform_view_screen.cpp
# covers: src/deluge/generative/perform/regenerate_page.*
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/firmware/adapter_timing.*
# covers: src/deluge/generative/firmware/entry_leave.cpp src/deluge/gui/context_menu/generative.cpp
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view_clip.cpp
# covers: src/deluge/gui/views/perform_view_song.cpp
start_state
hold entropy
include perform_open.txt
include perform_confirm.txt
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0

# ---- Acid techno has no sets: the row under REGENERATE is KEY / SCALE, no VARIANTS ----
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
expect no_event gen_variants_menu within=1
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 100

# ---- ROOTS DUB -> CUT: two detents from DROP ----
include perform_transition_menu.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre dub
expect event gen_genre_pick dub row=ROOTS DUB
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect ui=sound_editor
include perform_regen_transition.txt
encoder 3 2
include perform_regen_go.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_confirm dub style=2 result=0
expect event gen_transition_preload done error=0 parts=5 within=30
expect event gen_transition start style=2 within=90
expect event gen_transition phase=idle within=60

# ---- VARIANTS ▸ after REGENERATE: SKANK and DRUMS on RANDOM ----
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_variants_menu open sets=2
expect event gen_variants_row SKANK Random
expect event gen_variants_row DRUMS Random
# DRUMS' multi-select, from the top (a list remembers its row) on ONE DROP: untick it, then ROCKERS; each turn queues
mark now
encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 100
encoder 3 -5
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_variants_press set=drums allowed=6 outcome=0
expect event gen_variants set=drums allowed=6 shows=2 of 3
encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_variants_press set=drums allowed=4 outcome=0
expect event gen_variants set=drums allowed=4 shows=STEPPERS
# STEPPERS' tick is the last: it stays, and queues nothing
encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_variants set=drums allowed=4 shows=STEPPERS
expect no_event gen_variants_press within=1
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 200
expect event /^gen_status .*\|>VAR/ since=mark within=5
# No Drop yet: it waits
expect no_event gen_variants_landed within=4

# ---- A forced 1-bar Build: the ticks land on its Drop, which draws STEPPERS, the one ticked ----
mark now
pad 17 5 down
wait 50
pad 17 5 up
expect event gen_press force BUILD
expect event gen_section BUILD within=40
expect event gen_section GROOVE within=10
expect event /^gen_variants_landed step=\d+ drop=1 skank=sevenths:\d+ drums=steppers:4$/ since=mark within=5
expect event gen_h1 steps= within=20
hardware_only expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]/ within=20
expect event gen_h1 steps= within=20
# The row shows the tick
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_variants_row DRUMS STEPPERS
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 100
expect playing=true

button PLAY down
wait 50
button PLAY up
expect playing=false
include perform_leave.txt
expect ui_mode=0
release entropy
