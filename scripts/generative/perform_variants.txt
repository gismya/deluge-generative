# Perform's VARIANTS ▸ and a live variant switch landing on a Drop (#495, spec §3.6, §13.6, ADR 0008 "Drop changes and
# live steering") over the sysex test port: Acid techno shows no VARIANTS row; Acid cuts to ROOTS DUB, whose Perform
# menu has VARIANTS ▸ after REGENERATE, a row per set with the variant playing (SKANK, DRUMS, both live). DRUMS opens
# its list: a pick queues a switch for the next Drop (VAR in the status line), and a forced 1-bar Build lands it on the
# Drop, where the instance applies its genre copy again. gen_h1 reports around that Drop (H1, spec §20). Runs on a
# Deluge (deluge_probe.py session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the
# current clip a synth or kit clip.
#
# gen_perform_variants open sets=N, then gen_perform_variants_row <SET> <playing> live=<0|1> per set, as VARIANTS ▸
# opens. gen_press variant set=<name> variant=<name> outcome=<0 accepted>: a turn in a set's list.
# gen_variants_landed step=N drop=<1 on a Drop's first step> <set>=<variant>:<allowed> ...: the picks the piece plays
# moved on step N (a Drop change or a live switch). Pinned rolls (hold entropy): the Dub Recipe plays ONE DROP and
# 7THS, so a turn to STEPPERS is a switch.
# For figures: --events 'gen_perform_variants|gen_press variant|gen_variants_landed|gen_section|gen_h1|gen_h16'.
# covers: src/deluge/gui/menu_item/generative/perform_menu.* src/deluge/generative/perform/perform_steering.h
# covers: src/deluge/generative/perform/queue_text.* src/deluge/generative/core/variants.*
# covers: src/deluge/generative/core/generator_steering.cpp src/deluge/generative/core/genres/dub.cpp
# covers: src/deluge/generative/core/steering.h src/deluge/generative/firmware/perform_adapter.cpp
# covers: src/deluge/gui/l10n/english.json src/deluge/gui/l10n/g_english.cpp
start_state
hold entropy
include perform_open.txt
include perform_confirm.txt
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0

# ---- Acid techno has no sets: the row under REGENERATE is KEY / SCALE, no VARIANTS ----
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
expect no_event gen_perform_variants within=1
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 100

# ---- ROOTS DUB -> CUT: two detents from DROP ----
include perform_transition_menu.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre dub
expect event gen_genre_pick dub row=ROOTS DUB
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect ui=sound_editor
include perform_regen_transition.txt
encoder 3 2
include perform_regen_go.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_confirm dub style=2 result=0
expect event gen_transition_preload done error=0 parts=5 within=30
expect event gen_transition start style=2 within=90
expect event gen_transition phase=idle within=60

# ---- VARIANTS ▸ after REGENERATE: SKANK and DRUMS, live, on what plays ----
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_perform_variants open sets=2
expect event gen_perform_variants_row SKANK 7THS live=1
expect event gen_perform_variants_row DRUMS ONE DROP live=1
# DRUMS' list, on ONE DROP: two detents down are ROCKERS, then STEPPERS, each a press; the last one waits
encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 100
encoder 3 1
expect event gen_press variant set=drums variant=rockers outcome=0
encoder 3 1
expect event gen_press variant set=drums variant=steppers outcome=0
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 100
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 200
expect event /^gen_status .*\|>VAR/ within=5
# No Drop yet: it waits
expect no_event gen_variants_landed within=4

# ---- A forced 1-bar Build: the switch lands on its Drop ----
mark now
pad 17 5 down
wait 50
pad 17 5 up
expect event gen_press force BUILD
expect event gen_section BUILD within=40
expect event gen_section GROOVE within=10
expect event /^gen_variants_landed step=\d+ drop=1 skank=sevenths:\d+ drums=steppers:\d+$/ since=mark within=5
expect event gen_h1 steps= within=20
hardware_only expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]/ within=20
expect event gen_h1 steps= within=20
# The row shows STEPPERS now
include perform_transition_menu.txt
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_perform_variants_row DRUMS STEPPERS live=1
button BACK down
wait 50
button BACK up
wait 100
button BACK down
wait 50
button BACK up
wait 100
expect playing=true

button PLAY down
wait 50
button PLAY up
expect playing=false
include perform_leave.txt
expect ui_mode=0
release entropy
