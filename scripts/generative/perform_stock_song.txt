# The tick gate with a stock song (#251, from #242): session.cpp calls performAdapter.tick() only while Perform is
# active, which touches every song's tick path. Load the user's song Ttt (it has an arrangement), play it in session
# view, then in the arranger, and check Perform's tick stays off (no gen_h1, which only Perform's step decisions send).
# Listen on the device: Ttt plays as it does on stock firmware.
# needs: SONGS/Ttt.XML on the card (the desk Deluge's; delugemu's card has no songs), so check_all.py skips it
# covers: src/deluge/playback/mode/session.cpp src/deluge/playback/mode/arrangement.cpp
# covers: src/deluge/generative/firmware/adapter_timing.*
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/firmware/output_sink.cpp
start_state
load song SONGS/Ttt.XML
expect event song_loaded path=SONGS/Ttt.XML within=120
# A second start_state, the load song exception in scripts.md: Ttt opens in whatever view it was saved in.
start_state
expect playing=false

# SHIFT+SONG from the clip goes to Song view; PLAY there plays the session
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=session
expect ui_mode=0
wait 1000
button PLAY down
wait 50
button PLAY up
expect playing=true
expect no_event gen_h1 within=15
expect playing=true
button PLAY down
wait 50
button PLAY up
expect playing=false

# SONG in Song view goes to the arranger; PLAY there plays the arrangement from its start
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
expect ui=arranger within=5
wait 1000
button PLAY down
wait 50
button PLAY up
expect playing=true
expect no_event gen_h1 within=20
expect playing=true
button PLAY down
wait 50
button PLAY up
expect playing=false
