# hold entropy pins Perform's rolls (TestPort::pinnedEntropy()): from each hold on, fresh entropy comes from the same
# fixed sequence, so two entries, each after its own hold, roll the same Recipe and part presets and load the same
# sounds. Which sounds depends on the card, so the script compares the two entries' gen_sound events instead of naming
# them. Every entry rolls a fresh Recipe (spec §14.1): without the pin the second would load other sounds.
# covers: src/deluge/generative/core/draw.* src/deluge/generative/core/recipe.* src/deluge/generative/core/palette.*
# covers: src/deluge/generative/firmware/card_sounds.* src/deluge/generative/firmware/current_recipe.*
# covers: src/deluge/generative/firmware/entry_leave.cpp
start_state
hold entropy
include perform_open.txt
mark
include perform_confirm.txt
keep events /^gen_sound /
include perform_leave.txt

# ---- The hold again: the sequence starts over, so the second entry loads the same sounds ----
hold entropy
# Back in session view: SHIFT+SONG opens GENERATIVE (right after reaching session view it needs about 1 s)
sleep 1
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=context_menu
include perform_pick_acid.txt
mark
include perform_confirm.txt
expect events kept within=10
include perform_leave.txt
