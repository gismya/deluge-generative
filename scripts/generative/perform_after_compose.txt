# #388: does Perform entry's Generator land in SDRAM after a large Compose song, and does that slow step decisions
# (H1, spec §20)? Perform entry allocates its two Generator blocks with allocMaxSpeed while the old song is still
# loaded; a 20:00 Compose song can leave the internal heap nearly full (H19: about 89 K free).
# Two Perform entries on Acid techno, each played into a forced Groove for four gen_h1 reports (16 bars):
#   1. a normal entry, from start_state's blank song;
#   2. an entry right after a 20:00 Psytrance Compose song (the H19 build), with that song still current.
# gen_generator_heap gives each block's heap (internal or sdram), its address range and the internal heap's free
# bytes after the allocation; gen_mem before_entry the regions as entry starts. Compare H1's mean_ns between the two.
# Runs on a Deluge (deluge_probe.py session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped,
# the current clip a synth or kit clip.
# needs: a Deluge for its timings, and about 5 minutes in delugemu, so check_all.py skips it
# For figures: --events 'gen_generator_heap|gen_mem|gen_h1|gen_h16|gen_section|gen_compose'.
# covers: src/deluge/generative/firmware/entry_leave.cpp src/deluge/generative/firmware/compose_job.cpp
# covers: src/deluge/generative/firmware/adapter_timing.h src/deluge/generative/firmware/output_sink.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/gui/menu_item/generative/perform_menu.cpp
# covers: src/deluge/gui/views/perform_view_clip.cpp src/deluge/gui/views/perform_view_song.cpp
start_state
hold entropy

# ---- 1. A normal entry, from a blank song ----
mark now
include perform_open.txt
include perform_confirm.txt
expect event /^gen_generator_heap block=0 (internal|sdram) start=0x[0-9a-f]{8} end=0x[0-9a-f]{8} internal_free=\d+$/ since=mark
expect event /^gen_generator_heap block=1 (internal|sdram) start=/ since=mark
include perform_after_compose_groove.txt
include perform_leave.txt

# ---- 2. An entry after a 20:00 Compose song ----
compose 1200 psytrance
expect event /^gen_compose done ran=1 / within=600
expect ui=arranger within=10

button SESSION_VIEW up
expect ui=session within=10
wait 1000
mark now
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=context_menu
expect root_ui=session
include perform_pick_acid.txt
include perform_confirm.txt
expect event /^gen_generator_heap block=0 (internal|sdram) start=0x[0-9a-f]{8} end=0x[0-9a-f]{8} internal_free=\d+$/ since=mark
expect event /^gen_generator_heap block=1 (internal|sdram) start=/ since=mark
include perform_after_compose_groove.txt
include perform_leave.txt
