# Perform's transport (#108, spec §13.2, §12.4, §12.7, §13.7) over the sysex test port. Start as after boot:
# transport stopped, the current clip a synth or kit clip. The external-clock part sends MIDI clock from the host
# ("clock"), so run it in delugemu (test_port_emu.py) or on a Deluge whose USB MIDI port receives clock.
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view.h
# covers: src/deluge/gui/views/perform_view_internal.h src/deluge/generative/firmware/perform_adapter.*
# covers: src/deluge/generative/firmware/output_sink.cpp src/deluge/gui/views/perform_view_screen.cpp
# covers: src/deluge/generative/perform/clock_follow.h src/deluge/playback/
# covers: src/deluge/gui/views/perform_view_song.cpp src/deluge/generative/firmware/entry_leave.cpp
# covers: src/deluge/gui/context_menu/generative.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/gui/l10n/english.json src/deluge/gui/l10n/g_english.cpp
start_state
include perform_open.txt
include perform_confirm.txt

# ---- TAP and the tempo encoder (F4), stopped: the Recipe tempo, written to the song at once ----
# Taps 500 ms apart: 120 BPM on a Deluge. delugemu's audio clock doesn't keep to its millisecond one, so there the
# tapped tempo is only near it, and the script checks the popup, not the value.
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
wait 450
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
wait 450
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
expect event gen_tempo_popup Tempo:
expect event gen_tempo_popup Tempo:
# Once the popup is gone, the first turn shows the tempo and the next one moves it to the next whole BPM
wait 3000
encoder 2 1
wait 300
encoder 2 1
expect event gen_tempo_popup Tempo:
expect event gen_tempo_popup Tempo:
# LEARN + turn is swallowed: the clock-out scale can't change in Perform, and no popup
button LEARN down
wait 50
encoder 2 1
wait 300
button LEARN up
expect no_event gen_tempo_popup within=1
expect ui=generative_perform

# ---- SHIFT+PLAY starts in Groove, with no Intro ----
button SHIFT down
wait 50
button PLAY down
wait 50
button PLAY up
wait 50
button SHIFT up
expect playing=true
expect event gen_start master=0 to_step0=0 groove=1
expect event gen_section GROOVE 0 clip_pos=0 within=5

# TAP while playing glides over the Glide setting (1 bar, 2 s at 120): taps 600 ms apart, 100 BPM
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
wait 550
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
wait 550
button TAP_TEMPO down
wait 50
button TAP_TEMPO up
expect event gen_tempo_popup Tempo:
expect event gen_tempo_popup Tempo:
# SHIFT + turn: swing, which glides on step boundaries. The first turn shows it.
wait 3000
button SHIFT down
wait 50
encoder 2 1
wait 300
encoder 2 1
wait 300
encoder 2 1
wait 300
button SHIFT up
# within= counts from here, the 4 s of queued waits and turns above included
expect swing=2 within=10

# ---- End: a tap cancels it while queued ----
include perform_hold_end.txt
pad 17 0 down
wait 100
pad 17 0 up
expect event gen_end tap
# Techno's phrase is 8 bars, and the tap came in the first. Wait on the music, not the host's clock: gen_phrase comes
# once the Generator has decided the bar-8 line, just before it plays (an uncancelled Outro's gen_section came 70 ms
# later in delugemu). 19.2 s of music at the tapped 100 BPM, which delugemu under load plays more slowly than real
# time: twice that. gen_phrase is sent after the step is decided, so a tick late enough sends bar 8's Section first:
# the no_event counts from the mark at the tap, so an Outro either side of gen_phrase fails it.
mark
expect event gen_phrase bar=8 within=40
expect no_event gen_section OUTRO since=mark within=3
expect playing=true

# ---- End held 0.6 s: the Outro from the next phrase line, then the transport stops where it ends ----
# Armed just past bar 8, the Outro starts on bar 16: neither a phrase late nor at bar 8 (the tap cancelled that one;
# a bar-8 Outro would have ended the piece on bar 16). Same doubled windows.
include perform_hold_end.txt
expect event /^gen_section OUTRO 0 clip_pos=\S+ bar=16$/ within=40
expect event gen_piece_ended within=40
expect event playback_stop
expect playing=false

# ---- External clock (spec §12.4, §13.7) ----
# Live sound-control ramps (#106) cost delugemu's emulated CPU enough that its reading of the host's clock jitters
# (118-119 for 120 BPM); a Deluge has the headroom (spec §20 H2). This part tests the clock, not the sound, so it
# holds them.
hold ramps
# MIDI Start: step 0 at tick 0, the normal start, with the Intro
clock 120
midi fa
expect playing=true
expect event gen_start master=0 to_step0=0 groove=0
expect event gen_section INTRO 0 clip_pos=0 within=5
# The master owns tempo: a turn shows it marked EXT and changes nothing, and the Recipe tempo tracks it per bar
encoder 2 1
wait 300
encoder 2 1
expect event /^gen_tempo_popup .* EXT$/
# The tempo it reads off the host's clock is measured against delugemu's audio clock, which jitters under host load
# (118-123 over a minute). Each bar's reading is logged only when it changes, so the script waits for one within
# 1 BPM: the readings scatter around the master's tempo.
expect event /^gen_ext_tempo bpm=(119|120|121)$/ within=10
# A mid-play song position pointer, 1 beat in: note-offs, then silence to the next master bar line, 12 steps on
midi f2 04 00
expect event gen_jump spp master=96 to_line=288
expect no_event gen_section INTRO 0 within=3
expect playing=true
midi fc
expect playing=false

# Continue from 2 beats in: silent until the next master bar line, 8 steps on, then step 0 (a fresh piece)
midi f2 08 00
midi fb
expect playing=true
expect event gen_start master=192 to_step0=192 groove=0
# The stock clips agree: step 0 plays on their bar line
expect event gen_section INTRO 0 clip_pos=0 within=5
# A few bars, so the Recipe tempo has tracked the clock
sleep 6
midi fc
expect playing=false

# A song position pointer sent mid-step, 5 steps into a bar: the piece resumes on the master's bar line, so the next
# Section (the Outro End queues) starts there by the stock clips too
midi fa
expect event gen_section INTRO 0 clip_pos=0 within=5
include perform_hold_end.txt
wait 1310
midi f2 05 00
expect event gen_jump spp master=120 to_line=264
# 8 bars of the host's 120 BPM clock each (16 s), which a loaded host can send late. End armed in the first phrase:
# the Outro on bar 8, not a phrase late
expect event gen_section OUTRO 0 clip_pos=0 bar=8 within=40
expect event gen_piece_ended within=40
expect playing=false
clock off
release ramps

# Clock lost: the host's clock stops with no MIDI Stop. Stock MIDI clock has no timeout (only an analog clock in
# auto-start mode stops itself), so the transport stays on the lost clock, the piece held where it was, until PLAY
# stops it as stock does (spec §13.7)
clock 120
midi fa
expect playing=true
expect event gen_start master=0 to_step0=0 groove=0
expect event gen_section INTRO 0 clip_pos=0 within=5
clock off
sleep 2
expect playing=true
button PLAY down
wait 50
button PLAY up
expect playing=false

# Back on the internal clock, PLAY starts at the last tempo tracked: gen_start's tempo is the last gen_ext_tempo's
# (both 120 on a steady clock; delugemu's wobbles a BPM, so the script only checks the start)
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_start master=0 to_step0=0 groove=0 tempo=
button PLAY down
wait 50
button PLAY up
expect playing=false

# ---- Leave ----
include perform_leave.txt
