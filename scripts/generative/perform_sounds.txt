# Sounds (#104) over the sysex test port: Perform's entry loads every sound through its palette's fallback chain with
# its load-time settings. Runs on a Deluge (deluge_probe.py run) or in delugemu (test_port_emu.py). Start as after
# boot: transport stopped, the current clip a synth or kit clip.
# gen_sound PART LANE SOURCE mode=<PolyphonyMode> prio=<VoicePriority> PATH; synths: mode= prio= arp=<ArpMode> PATH porta=<param>
#   PolyphonyMode: 0 AUTO, 3 LEGATO, 4 CHOKE. VoicePriority: 0 LOW, 2 HIGH. ArpMode 0 OFF.
# covers: src/deluge/generative/firmware/card_sounds.* src/deluge/generative/firmware/perform_song.*
# covers: src/deluge/generative/firmware/entry_leave.cpp src/deluge/generative/core/palette.*
# covers: src/deluge/generative/core/palette_file.* src/deluge/generative/core/sound_settings.h
# covers: src/deluge/generative/firmware/sound_swaps.* src/deluge/generative/perform/sound_swap_lifecycle.h
# covers: src/deluge/generative/firmware/adapter_timing.h src/deluge/gui/views/perform_view.cpp
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/gui/menu_item/generative/perform_menu.cpp
# covers: src/deluge/gui/views/perform_view_clip.cpp src/deluge/gui/views/perform_view_song.cpp
# covers: src/deluge/generative/firmware/output_sink.cpp src/deluge/playback/playback_handler.cpp
# covers: src/deluge/generative/core/genres/techno.cpp src/deluge/generative/core/types.h
# covers: src/deluge/generative/core/lane_names.h
start_state
include perform_open.txt

# OK rolls the Recipe's sound picks from the card, then loads them in roster and lane order (spec §10.1-§10.6)
button 4 3 down
wait 50
button 4 3 up
expect event gen_sound DRUMS KICK pick mode=0 prio=2 SAMPLES/DRUMS/Kick/ within=20
expect event gen_sound DRUMS SNARE pick mode=0 prio=2 SAMPLES/DRUMS/Snare/ within=20
expect event gen_sound DRUMS RIM pick mode=0 prio=2 SAMPLES/DRUMS/Rim/ within=20
# The hat pair chokes, and only the hat pair
expect event gen_sound DRUMS CLOSED HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatC/ within=20
expect event gen_sound DRUMS OPEN HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatO/ within=20
# The section rules' lanes: the Build roll's Snare and the Phrase accent's Crash (spec §9.5)
expect event gen_sound DRUMS CRASH pick mode=0 prio=2 SAMPLES/DRUMS/Crash/ within=20
# The Loop Bass: LEGATO, high priority, the preset's arp cleared, portamento set (spec §10.6), at the preset's level
# gen_sound_level PART volume=<PartSpec .volume> lower|same|higher played=<the patcher's final volume> preset=<its value
# before the part's settings>: the level the Sound plays at, against its preset's (#228)
expect event gen_sound_level ACID volume=100 same within=20
expect event gen_sound ACID - pick mode=3 prio=2 arp=0 SYNTHS/ within=20
# The Drone: POLY, low priority, a drone pad by the Pad role (spec §10.6), at half its preset's knob: it plays lower
expect event gen_sound_level DRONE volume=50 lower within=20
expect event gen_sound DRONE - pick mode=1 prio=0 arp=0 SYNTHS/ within=20
# FX lanes: Riser from SAMPLES/FX/Riser or the pack preset, Fall from factory files, Impact one of its two pack presets
# (Boom or Wash, rolled per piece, spec §10.5). Riser and Impact are silent on a card without the pack. Low priority.
expect event gen_sound FX RISER within=20
expect event gen_sound FX FALL pick mode=0 prio=0 SAMPLES/Artists/Danny Taurus/MoDrama/Downer within=20
expect event /^gen_sound FX IMPACT (pick mode=0 prio=0 SYNTHS\/GENERATIVE\/FX\/GEN Impact (Boom|Wash)\.XML|silent mode=0 prio=0 -)$/ within=20
expect event gen_perform_entered within=20
expect root_ui=generative_perform within=10
expect ui=generative_perform
expect ui_mode=0
expect playing=false

# It plays: the Intro, then the first Groove
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0
expect event gen_h1 steps= within=15
expect event gen_h1 steps= within=15
expect playing=true

# PLAY stops; BACK held, then LEAVE, leaves (spec §13.8). It leaves during a held NEW SOUND preload (KIT, DRUMS
# held): the cancelled preload's end leaves the leave's working animation up, and the leave's end takes it down
# (spec §12.10)
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event playback_stop
hold preload
include perform_new_sound_drums.txt
expect event gen_press sound part=0 previous=0 result=0 working=1
expect event gen_sound_preload_held part=0 within=10
include perform_hold_back.txt

expect event /gen_left .* working=0/ within=15
