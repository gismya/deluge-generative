# The Perform menu (#114, spec §13.3, §13.5, §13.6, §7.6) over the sysex test port: SELECT push opens it (KEY /
# SCALE, GLIDE, SEED + TAKES; the transitions' rows are perform_transition.txt's), SCALE opens the key and scale
# editor, a key change queues to the next phrase line and SCALE blinks while it waits, the key that plays cancels it,
# a stop drops it, and the gold knobs stay on the macros under the menu (F16). Runs on a Deluge (deluge_probe.py
# session) or in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current clip a synth or kit
# clip.
#
# Menus remember their last row, so each row is chosen from the top (encoder 3 -5 first). With no transition to
# cancel, the rows are GENRE, REGENERATE, KEY / SCALE, GLIDE, SEED + TAKES. gen_menu PERFORM|KEY_SCALE:
# a menu opened. gen_key_set root= scale= stopped|queued|refused: the editor set the key (stopped: the Recipe's at
# once). gen_key root= scale= phrase_line=yes|no: a key change landed. gen_scale_led blink|off. gen_glide <n>: the
# Glide setting (0 Instant .. 5 1 phrase). gen_rings macros ok|stale: the knob rings show the macros' centres.
start_state
include perform_open.txt
include perform_confirm.txt

# ---- Stopped: SELECT push, KEY / SCALE ▸, then SCALE to MAJOR and KEY to C, each the Recipe's at once ----
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_menu PERFORM
expect ui=sound_editor
encoder 3 -5
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
encoder 3 -5
encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
encoder 3 -12
button BACK down
wait 50
button BACK up
wait 200
encoder 3 -5
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
encoder 3 -12
expect event gen_key_set root=0 scale=0 stopped
# Out of the editor and the menu
button BACK down
wait 50
button BACK up
wait 200
button BACK down
wait 50
button BACK up
wait 200
button BACK down
wait 50
button BACK up
expect ui=generative_perform

# The next start plays the key set while stopped (spec §13.5)
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_key root=0 scale=0 phrase_line=yes within=5

# ---- The gold knobs under the menu: F8's macros, their rings kept (F16, F15) ----
# F1, then F8: Energy / Density, whatever set an earlier script left on F8

wait 50
button 2 3 down
wait 50

button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_menu PERFORM
expect ui=sound_editor
# Opening runs stock's knob-ring refresh: the rings stay on the macros' centres

encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
button BACK down
wait 50
button BACK up
expect no_event gen_rings macros stale within=1
# The upper knob turns Energy, and its ring follows
encoder 4 4
expect event gen_macro_knob macro=0
expect event gen_rings macros ok
button BACK down
wait 50
button BACK up
expect ui=generative_perform

# ---- Page LEDs under the menu: CLIP lit on the CLIP page, AFFECT ENTIRE dark with a part locked (spec §13.5) ----
# gen_page_leds song= clip= keyboard= affect= cross=: the LEDs after Perform's per-frame check, on a change. Stock's

encoder 3 1
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
button BACK down
wait 50
button BACK up

button BACK down
wait 50

expect ui=generative_perform

# ---- Playing: SCALE opens the key and scale editor; KEY to D queues to the next phrase line ----

expect event gen_scale_led off
# D again, then out of the menu: it lands on a phrase line, and SCALE stops blinking
