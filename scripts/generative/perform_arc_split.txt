# The ARC page's split box (#545, spec §13.5): a phrase that a Section starts partway into shows both kinds. Dub's
# Break plays 7 bars and its 1-bar Build the phrase's last (its BREAK set's SEVEN; FIFTEEN, which entropy 7 draws,
# plays one more phrase of Break first), so the Break's last phrase is split: BR on top, BU below. NEXT names each Section as it comes: the
# Break on its line, the Build in the phrase's last bar, then the Drop. Start as after boot: transport stopped, the
# current clip a synth or kit clip.
#
# gen_arcpage: the ARC page's eight boxes (tag, * on a Drop, -- after a stop; a split box gives the Section its phrase
# starts with, then > and the one it ends with: BR>BU) | the next change. Dub at 160 BPM (next recipe), so a phrase
# takes 12 s.
# covers: src/deluge/generative/perform/arc_map.* src/deluge/generative/perform/oled_page.*
# covers: src/deluge/generative/perform/phrase_history.* src/deluge/generative/core/arc_plan.h
# covers: src/deluge/generative/core/generator_steering.cpp src/deluge/gui/views/perform_view_screen.cpp
# covers: src/deluge/generative/core/genres/dub.cpp src/deluge/generative/core/types.h
# covers: src/deluge/generative/perform/queue_text.* src/deluge/generative/perform/text.*
# covers: src/deluge/gui/views/perform_view.cpp src/deluge/gui/views/perform_view_song.cpp
# covers: src/deluge/generative/firmware/entry_leave.cpp src/deluge/generative/core/genres/techno.cpp
# covers: src/deluge/gui/context_menu/generative.cpp src/deluge/gui/menu_item/generative/perform_menu.cpp
# covers: src/deluge/gui/views/perform_view_clip.cpp
start_state
next recipe genre dub
next recipe seed 3
next recipe entropy 7
next recipe tempo 16000
include perform_open.txt
mark
include perform_confirm.txt
expect event gen_next_recipe taken since=mark
expect event gen_perform_entered dub since=mark

# ---- SHIFT+PLAY: Groove from the first bar, whole boxes ----
mark
button SHIFT down
wait 50
button PLAY down
wait 50
button PLAY up
wait 50
button SHIFT up
expect playing=true
expect event /^gen_arcpage GR GR GR / since=mark within=5

# ---- Force a Break: its last phrase's box is split, the Drop's after it; NEXT is the Break on its line ----
pad 17 6 down
wait 50
pad 17 6 up
expect event gen_press force BREAK
expect event /^gen_arcpage GR (BR )?BR>BU GR\*.*\|NEXT BREAK / within=5

# ---- In the split phrase, before its last bar: now's box is split, the Build is next ----
expect event /^gen_arcpage BR>BU GR\*.*\|NEXT BUILD / within=40

# ---- In the Build: the box stays split, the Drop is next ----
expect event /^gen_arcpage BR>BU GR\*.*\|NEXT DROP / within=15

# ---- The Drop: a whole Groove box ----
expect event /^gen_arcpage GR\*GR / within=5
include perform_leave.txt
