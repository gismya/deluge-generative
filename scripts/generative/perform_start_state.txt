# start_state releases every hold (#193) over the sysex test port: a script that stopped mid-hold leaves nothing held
# for the next one on the same load. Runs on a Deluge (deluge_probe.py run) or in delugemu (test_port_emu.py). Start
# as after boot: transport stopped, the current clip a synth or kit clip.
start_state

# ---- A script stops here, with card reads failing, the next preload parked and fill mode on ----
hold card_error
hold preload
hold midi_fill

# ---- The next script's start_state ("release all"): Perform's entry reads its sounds from the card ----
start_state
