# Ambient's genre changes (#125, spec §9.7, §21.3) over the sysex test port, in one boot: Acid techno into Ambient
# and back out in all three styles. AMBIENT is a GENRE row of its own (no parent), so picking it opens the confirm at
# once. Its transition entry is Intro, so the confirm asks AMBIENT? on FADE THROUGH: accepting with no turn plays
# style=1, and each other style into Ambient is one detent from it. Back to ACID, the confirm starts on BUILD INTO A
# DROP. In Ambient after the first change a forced Break keeps the Pad, Drone and bells sounding (gen_h16 voices=), and
# the last change out of it plays Ambient's 16-bar Build, sounding too. Runs on a Deluge (deluge_probe.py session) or
# in delugemu (test_port_emu.py). Start as after boot: transport stopped, the current clip a synth or kit clip.
#
# Order: Acid -> Ambient FADE (the highlight), a Break in Ambient, -> Acid CUT, -> Ambient BUILD, -> Acid FADE
# (Ambient's 2-phrase Outro), -> Ambient CUT, -> Acid BUILD (Ambient's Build). Ambient plays 60-90 BPM in 8-bar phrases,
# so a phrase is 21-32 s: its waits are long.
# gen_confirm <genre> style=<0 drop|1 fade|2 cut> result=<0 started>; gen_outputs_freed ... total=N counts the outgoing
# roster's Outputs: Acid techno's 4, Ambient's 6.
# For figures: --events 'gen_confirm|gen_transition|gen_section|gen_h1|gen_h16|gen_outputs_freed'.
# covers: src/deluge/generative/core/genres/ambient.cpp src/deluge/generative/core/genres/ambient_palette.cpp
# covers: src/deluge/generative/core/genre.* src/deluge/gui/context_menu/generative.*
# covers: src/deluge/generative/perform/genre_menu.* src/deluge/gui/menu_item/generative/genre_rows.*
# covers: src/deluge/generative/firmware/transitions.* src/deluge/generative/perform/transition_lifecycle.h
# covers: src/deluge/generative/firmware/retirements.* src/deluge/generative/firmware/adapter_timing.*
# covers: src/deluge/generative/firmware/output_sink.cpp

# covers: src/deluge/generative/core/types.h
start_state
hold entropy
include perform_open.txt
include perform_confirm.txt
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0

# ---- AMBIENT -> FADE THROUGH, the highlighted style: Acid techno's Outro, then Ambient's Intro at the seam ----
mark
include perform_transition_menu.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre ambient
expect event gen_genre_pick ambient row=AMBIENT
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect ui=context_menu
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_confirm ambient style=1 result=0
expect event gen_transition_preload done error=0 parts=6 within=30
expect event gen_transition start style=1 within=90
expect event /^gen_outputs_freed outputs=\d+ total=4$/ within=40
expect event gen_transition phase=idle within=10
expect playing=true

# ---- A Break in Ambient: forced, it lands on the next phrase line and keeps the Pad, Drone and bells sounding ----
pad 17 6 down
wait 50
pad 17 6 up
expect event gen_section BREAK within=90
# The H16 report comes every 4 bars, on the phrase grid: the one on the Break's line goes out just before its Section
# event and counts the Groove's voices, so the Break's own comes 4 bars later, 13-16 s at this tempo. within= is host
# time: under load delugemu runs slower than real time, so the wait has room for that.
expect event /^gen_h16 culls=\d+ culled=\S+ voices=[1-9]/ within=40
expect event gen_h1 steps= within=20
