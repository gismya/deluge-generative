# One Perform enter/play/leave cycle, for gen_emu.py leak (the memory check), which runs it N times on one boot and
# reads the gen_mem before_entry / after_leave events. Not a check on its own: check_all.py skips it.
# needs: gen_emu.py leak
start_state
include perform_open.txt
include perform_confirm.txt
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0
sleep 2
button PLAY down
wait 50
button PLAY up
expect playing=false
include perform_leave.txt
