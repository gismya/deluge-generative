# Sysex test port smoke run (#46). The same script runs in delugemu (test_port_emu.py) and on a Deluge
# (deluge_probe.py run). Start with the transport stopped and a song whose current clip is a synth or kit clip,
# such as the factory boot song.
expect playing=false

# SHIFT+SONG goes to Song view
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=session
expect ui_mode=0  # the zoom-out animation has finished

# CLIP goes back into the current clip
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect root_ui=instrument_clip

# A gold-knob turn shows its parameter's popup on the OLED
sleep 2
snapshot
encoder 5 -2
wait 300
expect oled_changed

# PLAY starts the transport, and stops it again
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event playback_start
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event playback_stop
