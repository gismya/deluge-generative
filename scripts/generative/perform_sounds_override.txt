# Sounds (#104) with a palette override file, in delugemu (test_port_emu.py --sd-master <card>). The card is the
# factory card plus tests/spec/generator/fixtures/palette_override/ (GENERATIVE/PALETTES/TECHNO.XML) and a folder
# SAMPLES/DRUMS/MyKicks holding "Mine 1.wav": the override points Kick at that folder, Rim at one exact file and ACID
# at one preset; every other lane keeps the built-in palette (spec §10.1, §10.2).
# needs: its own card (above), so check_all.py skips it
start_state
expect playing=false
# SHIFT+SONG from the clip goes to Song view
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=session
expect ui_mode=0

# SHIFT+SONG in Song view opens GENERATIVE over it
button SHIFT down
wait 100
button SESSION_VIEW down
wait 100
button SESSION_VIEW up
wait 100
button SHIFT up
expect ui=context_menu
expect root_ui=session

# PERFORM (SELECT push) opens PERFORM FROM full screen
button 4 3 down
wait 50
button 4 3 up
expect ui=sound_editor

# GENRE ▸ opens on TECHNO ▸, which opens on ACID (today's Techno, the default genre), which asks DISCARD SONG?
button 4 3 down
wait 50
button 4 3 up
wait 300
button 4 3 down
wait 50
button 4 3 up
wait 300
button 4 3 down
wait 50
button 4 3 up
expect ui=context_menu

# OK rolls the Recipe's sound picks from the card, then loads them in roster and lane order (spec §10.1-§10.6)
button 4 3 down
wait 50
button 4 3 up
expect event gen_palette_override techno read within=20
expect event gen_sound DRUMS KICK pick mode=0 prio=2 SAMPLES/DRUMS/MyKicks/Mine 1.wav within=20
expect event gen_sound DRUMS SNARE within=20
expect event gen_sound DRUMS RIM pick mode=0 prio=2 SAMPLES/DRUMS/Rim/808 Rim.wav within=20
# The hat pair chokes, and only the hat pair
expect event gen_sound DRUMS CLOSED HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatC/ within=20
expect event gen_sound DRUMS OPEN HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatO/ within=20
# The roll's Snare and the accent's Crash
expect event gen_sound DRUMS CRASH within=20
# The Loop Bass: LEGATO, high priority, the preset's arp cleared, portamento set (spec §10.6)
expect event gen_sound ACID - pick mode=3 prio=2 arp=0 SYNTHS/047 Basic Dirty Bass.XML porta= within=20
# FX lanes: Riser from SAMPLES/FX/Riser or the pack preset (silent on a card without the pack), Fall and Impact
# from factory files. Low priority.
expect event gen_sound FX RISER within=20
expect event gen_sound FX FALL pick mode=0 prio=0 SAMPLES/Artists/Danny Taurus/MoDrama/Downer within=20
expect event gen_sound FX IMPACT pick mode=0 prio=0 SAMPLES/ within=20
expect event gen_perform_entered within=20
expect root_ui=generative_perform within=10
expect ui=generative_perform
expect ui_mode=0
expect playing=false

# It plays: the Intro, then the first Groove
button PLAY down
wait 50
button PLAY up
expect playing=true
expect event gen_section INTRO 0
expect event gen_h1 within=15
expect event gen_h1 within=15
expect playing=true

# PLAY stops; BACK held, then LEAVE, leaves (spec §13.8)
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event playback_stop
button BACK down
wait 700
button BACK up
expect ui=context_menu
button 4 3 down
wait 50
button 4 3 up
expect root_ui=session within=15
expect ui_mode=0
expect playing=false
