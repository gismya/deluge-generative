# Sounds (#104) with a palette override file, in delugemu (test_port_emu.py --sd-master <card>). The card is the
# factory card plus tests/spec/generator/fixtures/palette_override/ (GENERATIVE/PALETTES/ACID TECHNO.XML and MELODIC
# TECHNO.XML) and a folder SAMPLES/DRUMS/MyKicks holding "Mine 1.wav": Acid's override points Kick at that folder, Rim
# at one exact file and ACID at one preset, with its own control ranges; every other lane keeps the built-in palette
# (spec §10.1, §10.2, §11.1). A genre change to Melodic then stages Melodic's override, whose BASS is one preset (spec
# §13.6), and a re-roll of the BASS after the commit reads it too.
# needs: its own card (above), so check_all.py skips it
start_state
include perform_open.txt

# OK rolls the Recipe's sound picks from the card, then loads them in roster and lane order (spec §10.1-§10.6)
button 4 3 down
wait 50
button 4 3 up
expect event gen_palette_override techno-acid read within=20
expect event gen_sound DRUMS KICK pick mode=0 prio=2 SAMPLES/DRUMS/MyKicks/Mine 1.wav within=20
expect event gen_sound DRUMS SNARE within=20
expect event gen_sound DRUMS RIM pick mode=0 prio=2 SAMPLES/DRUMS/Rim/808 Rim.wav within=20
# The hat pair chokes, and only the hat pair
expect event gen_sound DRUMS CLOSED HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatC/ within=20
expect event gen_sound DRUMS OPEN HAT pick mode=4 prio=2 SAMPLES/DRUMS/HatO/ within=20
# The roll's Snare and the accent's Crash
expect event gen_sound DRUMS CRASH within=20
# The Loop Bass: LEGATO, high priority, the preset's arp cleared, portamento set (spec §10.6)
expect event gen_sound ACID - pick mode=3 prio=2 arp=0 SYNTHS/047 Basic Dirty Bass.XML porta= within=20
# The Drone: POLY, low priority, a drone pad by the Pad role (spec §10.6)
expect event gen_sound DRONE - pick mode=1 prio=0 arp=0 SYNTHS/ within=20
# FX lanes: Riser from SAMPLES/FX/Riser or the pack preset (silent on a card without the pack), Fall and Impact
# from factory files. Low priority.
expect event gen_sound FX RISER within=20
expect event gen_sound FX FALL pick mode=0 prio=0 SAMPLES/Artists/Danny Taurus/MoDrama/Downer within=20
expect event gen_sound FX IMPACT pick mode=0 prio=0 SAMPLES/ within=20
expect event gen_perform_entered within=20
expect root_ui=generative_perform within=10
expect ui=generative_perform
expect ui_mode=0
expect playing=false

# It plays: the Intro, then the first Groove
button PLAY down
wait 50
button PLAY up
expect playing=true
# ACID's preset sets its own ranges (brightness="200" drive="0"): its LPF moves ±16, twice the default ±8, and Drive
# binds nothing on it. No other set has ranges of its own, so this is the only such event
expect event /gen_controls ranges part=1 lane=- BRI=\d+:-16\.\.16 DEC=\d+:-6\.\.6 DRI=0 SPA=/ within=15
expect event gen_controls sets=13
expect event gen_section INTRO 0
expect event gen_h1 within=15
expect event gen_h1 within=15
expect playing=true

# ---- TECHNO ▸ MELODIC, HARD CUT AT PHRASE: the incoming picks come from Melodic's override, staged with the preload
# (Acid's has no BASS, and no Techno palette lists 001 Sync Bass) ----
include perform_transition_menu.txt
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre techno-melodic
expect event gen_genre_pick techno-melodic row=TECHNO
button SELECT_ENC down
wait 50
button SELECT_ENC up
wait 200
pick genre techno-melodic
expect event gen_genre_pick techno-melodic row=MELODIC
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect ui=context_menu
encoder 3 2
button SELECT_ENC down
wait 50
button SELECT_ENC up
expect event gen_confirm techno-melodic style=2 result=0
mark
expect event gen_palette_override techno-melodic read within=20
expect event /^gen_sound BASS - pick .*SYNTHS.001 Sync Bass\.XML/ since=mark within=20
expect event gen_transition_preload done error=0 parts=5 within=20
expect event gen_transition start style=2 within=40
# The incoming BASS binds with the ranges of Melodic's override (thin="50" space="0"): the HPF moves 0..+10, half the
# default, and Space binds nothing; Brightness keeps its default ±8
expect event /gen_controls ranges part=1 lane=- BRI=\d+:-8\.\.8 .*SPA=0 .*THI=1:0\.\.10/ since=mark within=40
expect event gen_transition phase=idle within=40
expect playing=true

# ---- A re-roll after the commit reads Melodic's override, not Acid's: lock the BASS (double tap its row on the SONG
# page), then NEW SOUND on the CLIP page ----
pad 9 6 down
wait 50
pad 9 6 up
wait 50
pad 9 6 down
wait 50
pad 9 6 up
expect event gen_focus part=1 locked=1
button CLIP_VIEW down
wait 50
button CLIP_VIEW up
expect event gen_page CLIP
expect event gen_partpage BASS
mark
pad 1 6 down
wait 50
pad 1 6 up
expect event gen_sound_press part=1 previous=0
expect event /^gen_sound BASS - pick .*SYNTHS.001 Sync Bass\.XML/ within=10
expect event gen_sound_preload done part=1 within=10
expect event gen_sound_swap part=1 line=1 within=40
# The re-rolled BASS binds again with its entry's ranges
expect event /gen_controls ranges part=1 lane=- BRI=\d+:-8\.\.8 .*SPA=0 .*THI=1:0\.\.10/ since=mark within=10

# PLAY stops, then leave (spec §13.8)
button PLAY down
wait 50
button PLAY up
expect playing=false
expect event playback_stop
include perform_leave.txt
